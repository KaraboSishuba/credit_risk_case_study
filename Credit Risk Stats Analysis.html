credit_summary <- customer_data %>%
  group_by(balance_tier) %>%
  summarise(
    count = n(),
    mean_credit = mean(credit_limit, na.rm = TRUE),
    sd_credit = sd(credit_limit, na.rm = TRUE),
    median_credit = median(credit_limit, na.rm = TRUE)
  )
